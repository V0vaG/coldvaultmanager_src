דוגמת פקודה (bash)

# --- variables you can adjust ---
HOST="http://127.0.0.1:5000"
TOKEN="{{ s.api_token }}"   # לחץ "העתק" למעלה והדבק כאן אם תרצה
FILE="/path/to/firmware.ino.bin"

# multipart/form-data upload:
curl -X POST "$HOST/api/upload_firmware" \
  -H "Authorization: Bearer $TOKEN" \
  -F "file=@${FILE}"
